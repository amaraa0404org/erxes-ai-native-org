<objective>
Ship the providers module — the resolver-level CRUD path for `ai_providers` plus the rate-limited Test Connection action. This is where D-06 (encrypt-at-resolver-boundary) is realized: the mutation accepts a plaintext `apiKey` in its input shape, encrypts via `encryptSecret(apiKey, subdomain)` BEFORE the doc reaches the Mongoose model, and discards the plaintext local. The GraphQL response carries a `hasApiKey: boolean` only — the value never round-trips back to any client.

Purpose: Realize PROV-01 (add provider, 6 kinds), PROV-02 (AES-256-GCM encryption + never expose), PROV-03 (edit without re-entering — rotate-only mode), PROV-04 (remove), PROV-05 (Test Connection within 10s). Also implement D-07 (Test Connection rate limit) and D-06b (no key in any response).

Output: Six TS files under `backend/plugins/ai_api/src/modules/providers/` + a comprehensive test suite mocking models + Redis + the PROVIDER_REGISTRY adapter calls. The Apollo + tRPC surface that exposes these resolvers lives in Plan 08; this plan ships pure module logic that Plan 08 wires.

      import { encryptSecret, decryptSecret, PROVIDER_REGISTRY, ProviderConstructorArgs, ILLMProvider } from 'erxes-api-shared/ai';
      import type { ProviderKind } from 'erxes-api-shared/ai';
      import type { IProvider, IProviderDocument } from '~/models/@types';

      export const encryptApiKeyAtBoundary = (
        doc: { apiKey?: string | null } & Partial<IProvider>,
        subdomain: string,
      ): Partial<IProvider> => {
        const { apiKey, ...rest } = doc;
        if (typeof apiKey === 'string' && apiKey.length > 0) {
          return { ...rest, encryptedApiKey: encryptSecret(apiKey, subdomain) };
        }
        return rest;
      };

      export const resolveProvider = (
        provider: IProviderDocument,
        subdomain: string,
      ): ILLMProvider => {
        const factory = PROVIDER_REGISTRY[provider.kind as ProviderKind];
        if (!factory) {
          throw new Error(`AI provider kind '${provider.kind}' is not registered`);
        }
        const apiKey = decryptSecret(provider.encryptedApiKey, subdomain);  // throws SecretRotationRequiredError if rotation
        const args: ProviderConstructorArgs = {
          apiKey,
          baseUrl: provider.baseUrl ?? undefined,
          subdomain,
          config: provider.config ?? undefined,
        };
        return factory(args);
      };

      export const sanitizeProviderRow = (row: IProviderDocument | IProvider) => {
        const { encryptedApiKey, ...rest } = row.toObject ? row.toObject() : row;
        return { ...rest, hasApiKey: !!encryptedApiKey };
      };
      ```

            userId: ctx.user._id,

      import { redis } from 'erxes-api-shared/utils';
      import { SecretRotationRequiredError } from 'erxes-api-shared/ai';
      import type { IModels } from '~/connectionResolvers';
      import { resolveProvider } from './service';

      const RATE_LIMIT_PER_HOUR = 10;

      export interface TestConnectionResult {
        ok: boolean;
        latencyMs: number;
        modelsAvailable: number;
        errorCode?: string;
      }

      export const runTestConnection = async (args: {
        providerId: string;
        userId: string;
        subdomain: string;
        models: IModels;
      }): Promise<TestConnectionResult> => {
        const { providerId, userId, subdomain, models } = args;

        // D-07 Rate limit: 10/hour per user via Redis incr+expire
        const hourKey = new Date().toISOString().slice(0, 13).replace(/:/g, '-');
        const rateKey = `ai:test:${subdomain}:${userId}:${hourKey}`;
        const count = await redis.incr(rateKey);
        if (count === 1) await redis.expire(rateKey, 3600);
        if (count > RATE_LIMIT_PER_HOUR) {
          await models.AiProviderTestAudit.recordAttempt({
            subdomain, userId, providerId,
            attemptedAt: new Date(), succeeded: false, errorCode: 'RATE_LIMITED',
          });
          throw new Error(
            `Test Connection rate limit exceeded — ${RATE_LIMIT_PER_HOUR} attempts per hour per user. Try again at the next hour.`,
          );
        }

        const provider = await models.AiProviders.getProvider(providerId);

        // Resolve adapter — decryptSecret may throw SecretRotationRequiredError per PROV-06
        let adapter;
        try {
          adapter = resolveProvider(provider, subdomain);
        } catch (e) {
          await models.AiProviderTestAudit.recordAttempt({
            subdomain, userId, providerId,
            attemptedAt: new Date(), succeeded: false,
            errorCode: e instanceof SecretRotationRequiredError ? 'SECRET_ROTATION' : 'PROVIDER_RESOLVE_FAILED',
          });
          throw e;
        }

        // Adapter's testConnection sanitizes its own errors (P1 acceptance gate in Plan 03/04)
        let result: TestConnectionResult;
        try {
          if (typeof (adapter as any).testConnection !== 'function') {
            throw new Error(`Provider adapter for kind '${provider.kind}' does not implement testConnection()`);
          }
          result = await (adapter as any).testConnection();
        } catch (e) {
          const errMessage = e instanceof Error ? e.message : 'unknown error';
          result = { ok: false, latencyMs: 0, modelsAvailable: 0, errorCode: errMessage.slice(0, 200) };
        }

        await models.AiProviderTestAudit.recordAttempt({
          subdomain, userId, providerId,
          attemptedAt: new Date(),
          succeeded: result.ok,
          latencyMs: result.latencyMs,
          errorCode: result.errorCode,
        });

        return result;
      };
      ```
