Ship the models module — the read-only path that surfaces "available models per configured provider" for MODL-01. The `ai_models` collection is the cache; this plan ships the read + sync logic.

`AiModel` GraphQL type (brief §5.2 — not in the brief verbatim, but extension type for MODL-01):
```graphql
type AiModel {
  _id: ID                   # null if live-only (not yet synced to ai_models)
  providerId: ID!
  modelId: String!
  capabilities: [String!]!
  contextWindow: Int!
  inputCostPer1M: Float
  outputCostPer1M: Float
  enabled: Boolean!
}
```

Merge strategy:
- Fetch persisted rows: `await models.AiModels.listForProvider(providerId)` → keyed by `modelId`
- Fetch live: `await adapter.listModels()` → keyed by `id`
- For each persisted row: emit it as-is (admin overrides for MODL-03 land in Phase 1.3 — Phase 1.2 just plumbs through whatever's stored)
- For each live entry NOT in persisted set: emit as `{ _id: null, providerId, modelId: live.id, capabilities, contextWindow, inputCostPer1M: 0, outputCostPer1M: 0, enabled: true }` — `_id: null` flags "this is live data, not yet a Mongo row"
- Plan 08 hands the union to GraphQL with the schema above

NOTE: this plan does NOT call the adapter unless asked to — the `aiModels(providerId)` query has an optional `refresh: boolean` parameter (default `false`). When `refresh: true`, the resolver calls the adapter AND upserts results to `ai_models` (caches them). When `false`, only `ai_models` rows are returned. This avoids hammering provider APIs on every page load.

Empty-state behavior: if `ai_models` has zero rows for the provider AND `refresh: false`, the resolver auto-promotes to `refresh: true` on the first call (lazy seeding) — Plan 08 documents this in the GraphQL schema description.

      ```typescript
      import { resolveProvider } from '~/modules/providers/service';
      import type { IModels } from '~/connectionResolvers';
      import type { IModelDocument } from '~/models/@types';
      import type { ModelInfo } from 'erxes-api-shared/ai';

      export interface MergedModel {
        _id: string | null;          // null if live-only (not yet persisted)
        providerId: string;
        modelId: string;
        capabilities: string[];
        contextWindow: number;
        inputCostPer1M: number;
        outputCostPer1M: number;
        enabled: boolean;
      }

      export const listModelsForProvider = async (args: {
        providerId: string;
        subdomain: string;
        models: IModels;
        refresh?: boolean;
      }): Promise<MergedModel[]> => {
        const { providerId, subdomain, models, refresh = false } = args;
        const persisted = await models.AiModels.listForProvider(providerId);
        const persistedById = new Map(persisted.map(r => [r.modelId, r]));

        const shouldQueryLive = refresh || persisted.length === 0;
        let live: ModelInfo[] = [];
        if (shouldQueryLive) {
          try {
            const provider = await models.AiProviders.getProvider(providerId);
            const adapter = resolveProvider(provider, subdomain);
            live = await adapter.listModels();
          } catch (e) {
            // Adapter failure non-fatal — return persisted-only (UI shows what we have)
            // SecretRotationRequiredError still bubbles via aiProvidersTest path; listing is best-effort
            // eslint-disable-next-line no-console
            console.warn(`listModelsForProvider: adapter listModels failed for provider ${providerId}: ${e instanceof Error ? e.message : 'unknown'}`);
          }

          if (refresh && live.length) {
            // Upsert live results into ai_models — fire-and-forget OK since we'll return merged below
            await syncModelsForProvider({ providerId, subdomain, models, _liveOverride: live });
          }
        }

        const merged: MergedModel[] = [];
        const liveById = new Map(live.map(l => [l.id, l]));

        // Emit persisted (with override precedence)
        for (const row of persisted) {
          merged.push({
            _id: row._id,
            providerId: row.providerId,
            modelId: row.modelId,
            capabilities: row.capabilities,
            contextWindow: row.contextWindow,
            inputCostPer1M: row.inputCostPer1M ?? 0,
            outputCostPer1M: row.outputCostPer1M ?? 0,
            enabled: row.enabled,
          });
        }
        // Emit live-only (not in persisted)
        for (const l of live) {
          if (!persistedById.has(l.id)) {
            merged.push({
              _id: null,
              providerId,
              modelId: l.id,
              capabilities: l.capabilities,
              contextWindow: l.contextWindow,
              inputCostPer1M: 0,
              outputCostPer1M: 0,
              enabled: true,
            });
          }
        }

        return merged.sort((a, b) => a.modelId.localeCompare(b.modelId));
      };

      export const syncModelsForProvider = async (args: {
        providerId: string;
        subdomain: string;
        models: IModels;
        _liveOverride?: ModelInfo[];  // internal: pass-through to avoid double-listing
      }): Promise<{ synced: number; providerId: string }> => {
        const { providerId, subdomain, models, _liveOverride } = args;
        let live: ModelInfo[];
        if (_liveOverride) {
          live = _liveOverride;
        } else {
          const provider = await models.AiProviders.getProvider(providerId);
          const adapter = resolveProvider(provider, subdomain);
          live = await adapter.listModels();
        }

        let synced = 0;
        for (const l of live) {
          await models.AiModels.updateOne(
            { providerId, modelId: l.id },
            {
              $set: {
                capabilities: l.capabilities,
                contextWindow: l.contextWindow,
                enabled: true,
              },
              $setOnInsert: {
                providerId,
                modelId: l.id,
                inputCostPer1M: 0,
                outputCostPer1M: 0,
              },
            },
            { upsert: true },
          );
          synced++;
        }
        return { synced, providerId };
      };
      ```

      ```typescript
      import { Resolver } from 'erxes-api-shared/core-types';
      import { IContext } from '~/connectionResolvers';
      import { listModelsForProvider } from './service';
      import { aiModelsInputSchema } from './types';

      export const modelQueries: Record<string, Resolver> = {
        async aiModels(_root, args: { providerId?: string; refresh?: boolean }, ctx: IContext) {
          await ctx.checkPermission('aiModelsView');
          const parsed = aiModelsInputSchema.parse(args);

          if (parsed.providerId) {
            return listModelsForProvider({
              providerId: parsed.providerId,
              subdomain: ctx.subdomain,
              models: ctx.models,
              refresh: parsed.refresh,
            });
          }

          // All providers: aggregate without remote refresh
          const providers = await ctx.models.AiProviders.find({}).lean();
          const all = await Promise.all(
            providers.map((p) =>
              listModelsForProvider({
                providerId: p._id,
                subdomain: ctx.subdomain,
                models: ctx.models,
                refresh: false,
              }),
            ),
          );
          return all.flat();
        },
      };
      ```
