Estimator (P7 + P8 mitigation):
```typescript
estimateCostCents({ providerKind, model, inputTokens, toolDefsOverheadTokens = 0, expectedOutputTokens = 1024 }) {
  const key = `${providerKind}:${model}` as CostTableKey;
  const entry = COST_TABLE[key];
  if (!entry) {
    // Unknown model — return conservative estimate (4096 ctx model with caller-set output)
    const drift = DRIFT_MULTIPLIER[providerKind] ?? 1.2;
    const inCents = ((inputTokens + toolDefsOverheadTokens) / 1_000_000) * 100 * 5.0 * drift;
    const outCents = (expectedOutputTokens / 1_000_000) * 100 * 15.0 * drift;
    return Math.ceil(inCents + outCents);
  }
  const drift = DRIFT_MULTIPLIER[providerKind];
  const inCents = ((inputTokens + toolDefsOverheadTokens) / 1_000_000) * 100 * entry.inputCostPer1M * drift;
  const outCents = (expectedOutputTokens / 1_000_000) * 100 * entry.outputCostPer1M * drift;
  return Math.ceil(inCents + outCents);
}
```
Note: COST_TABLE values are in USD per 1M tokens; multiplying by 100 converts to cents. Math.ceil ensures we never under-estimate (better to pre-check generously and reconcile down post-call than risk overage). `expectedOutputTokens` defaults to 1024 — Phase 1.3 sets this per call (often `maxOutputTokens` from the chat request).

D-02c alert thresholds — pubsub payload:
```typescript
{

  currentSpend: 412,          // cents
  cap: 500,
  monthlyKey: '202605',
  emittedAt: new Date().toISOString(),
}
```
Threshold "crossing" detection: BEFORE reservation `currentSpend < threshold * cap`, AFTER `currentSpend >= threshold * cap`. Each threshold emits exactly once per month per scope (suppress duplicate emissions via a Redis SET key `ai:budget:threshold:fired:{subdomain}:{scope}:{scopeId}:{monthlyKey}` with the crossed fractions stored as SADD members).

Resolution order (scope precedence) — NOT in scope for Phase 1.2:
The brief implies that user-level caps inherit from team caps inherit from workspace caps. Phase 1.2 ships the storage and the per-scope check; the *resolution* logic (which scope applies to a given invocation) is Phase 1.3 (LLM call site). Phase 1.2's `checkBudget` accepts a single (scope, scopeId) tuple and is called by Phase 1.3 with the already-resolved scope.

      ```typescript
      import { redis } from 'erxes-api-shared/utils';

      export type BudgetScope = 'workspace' | 'team' | 'user' | 'agent';

      export const monthlyKey = (date: Date = new Date()): string =>
        `${date.getUTCFullYear()}${String(date.getUTCMonth() + 1).padStart(2, '0')}`;

      export const secondsToEndOfMonth = (date: Date = new Date()): number => {
        const next = new Date(Date.UTC(date.getUTCFullYear(), date.getUTCMonth() + 1, 1, 0, 0, 0));
        return Math.ceil((next.getTime() - date.getTime()) / 1000);
      };

      export const bucketKey = (args: { subdomain: string; scope: BudgetScope; scopeId: string; monthKey?: string }): string =>
        `ai:budget:${args.subdomain}:${args.scope}:${args.scopeId}:${args.monthKey ?? monthlyKey()}`;

      export const thresholdFiredKey = (args: { subdomain: string; scope: BudgetScope; scopeId: string; monthKey?: string }): string =>
        `ai:budget:threshold:fired:${args.subdomain}:${args.scope}:${args.scopeId}:${args.monthKey ?? monthlyKey()}`;

      export const thresholdChannel = (args: { subdomain: string; scope: BudgetScope; scopeId: string }): string =>
        `ai:budget:threshold:${args.subdomain}:${args.scope}:${args.scopeId}`;

      export const getCurrentSpendCents = async (args: { subdomain: string; scope: BudgetScope; scopeId: string }): Promise<number> => {
        const v = await redis.get(bucketKey(args));
        return Number(v) || 0;
      };

      export const reconcileBudget = async (args: { subdomain: string; scope: BudgetScope; scopeId: string; estimatedCents: number; actualCents: number }): Promise<void> => {
        const k = bucketKey(args);
        const delta = args.actualCents - args.estimatedCents;
        if (delta !== 0) await redis.incrby(k, delta);
        await redis.expire(k, secondsToEndOfMonth());
      };

      export const emitThresholdCrossed = async (args: {
        subdomain: string;
        scope: BudgetScope;
        scopeId: string;
        currentSpend: number;        // after-reservation cents
        previousSpend: number;       // before-reservation cents
        cap: number;
        alertThresholds: number[];
      }): Promise<void> => {
        const { subdomain, scope, scopeId, currentSpend, previousSpend, cap, alertThresholds } = args;
        if (cap <= 0) return; // $0 cap is XCUT-08 — no thresholds to cross
        const firedSetKey = thresholdFiredKey({ subdomain, scope, scopeId });
        const channel = thresholdChannel({ subdomain, scope, scopeId });
        for (const t of alertThresholds) {
          const crossed = previousSpend / cap < t && currentSpend / cap >= t;
          if (!crossed) continue;
          const added = await redis.sadd(firedSetKey, String(t));
          if (added === 1) {
            await redis.publish(channel, JSON.stringify({
              threshold: t,
              currentSpend,
              cap,
              monthlyKey: monthlyKey(),
              emittedAt: new Date().toISOString(),
            }));
          }
        }
        await redis.expire(firedSetKey, secondsToEndOfMonth());
      };
      ```

      ```typescript
      import { redis } from 'erxes-api-shared/utils';
      import { BudgetExceededError } from 'erxes-api-shared/ai';
      import type { IModels } from '~/connectionResolvers';
      import {
        BudgetScope,
        bucketKey,
        secondsToEndOfMonth,
        monthlyKey,
        emitThresholdCrossed,
      } from './service';

      export interface CheckBudgetArgs {
        subdomain: string;
        scope: BudgetScope;
        scopeId: string;
        estimatedCents: number;
        models: IModels;
      }

      export interface CheckBudgetResult {
        reservedCents: number;
        currentSpend: number;
        cap: number;
      }

      export const checkBudget = async (args: CheckBudgetArgs): Promise<CheckBudgetResult> => {
        const { subdomain, scope, scopeId, estimatedCents, models } = args;
        const monthKey = monthlyKey();

        // Resolve cap
        const budget = await models.AiBudgets.getBudget(scope, scopeId, monthKey);
        if (!budget) {
          // No cap configured for this scope this month — unlimited.
          // (AI_BUDGET_DEFAULT_USD env wiring lives in Phase 1.3 when scope-resolution chain ships.)
          return { reservedCents: 0, currentSpend: 0, cap: Number.POSITIVE_INFINITY };
        }

        const capCents = Math.round(budget.monthlyUsdCap * 100);
        const key = bucketKey({ subdomain, scope, scopeId, monthKey });

        const newSpend = await redis.incrby(key, estimatedCents);
        if (newSpend === estimatedCents) {
          await redis.expire(key, secondsToEndOfMonth());
        }

        if (newSpend > capCents) {
          await redis.decrby(key, estimatedCents);
          throw new BudgetExceededError({
            scope,
            scopeId,
            currentSpend: newSpend - estimatedCents,
            cap: capCents,
            suggestionUrl: `/settings/ai/budgets?scope=${scope}&scopeId=${encodeURIComponent(scopeId)}`,
          });
        }

        // Fire-and-forget threshold emission (don't await — pre-check latency must stay tight)
        void emitThresholdCrossed({
          subdomain,
          scope,
          scopeId,
          currentSpend: newSpend,
          previousSpend: newSpend - estimatedCents,
          cap: capCents,
          alertThresholds: budget.alertThresholds ?? [0.8],
        }).catch((e) => {
          // eslint-disable-next-line no-console
          console.warn(`emitThresholdCrossed failed for ${scope}:${scopeId}:`, e instanceof Error ? e.message : 'unknown');
        });

        return { reservedCents: estimatedCents, currentSpend: newSpend, cap: capCents };
      };

      export const releaseBudget = async (args: {
        subdomain: string;
        scope: BudgetScope;
        scopeId: string;
        estimatedCents: number;
      ```
