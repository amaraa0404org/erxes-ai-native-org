Rotation script algorithm (D-03c):
1. Parse argv: `--dry-run`, `--subdomain <slug>` (optional filter), `--verbose`
2. Read every distinct subdomain from Mongo (query all `*_ai_providers` collections, or use the existing service-discovery list of registered tenants — query simplicity wins: iterate `mongoose.connection.db.listCollections()` and grep for `_ai_providers` suffix, OR use `generateModels` with each subdomain explicitly — pick the simpler `listCollections` path)
3. For each subdomain, for each provider doc:
   - Try `decryptSecret(provider.encryptedApiKey, subdomain)` — wraps `parseSecretVersions` internally, so it auto-tries newest-first and falls back to older versions
   - If decrypt succeeds: re-encrypt via `encryptSecret(plaintext, subdomain)` (always uses NEWEST version)
   - Compare the new ciphertext to the existing one (envelope bytes-equality) — if equal → already-newest → skip (idempotent guarantee)
   - If different → write back via `models.AiProviders.updateOne({_id: provider._id}, { $set: { encryptedApiKey: newCiphertext } })`
   - If `--dry-run`: log the action but skip the write
4. On decrypt failure (SecretRotationRequiredError) → log "subdomain=X providerId=Y NEEDS_RESET (decryption failed under any active version)" and continue (do not crash the entire rotation — surface the un-recoverable providers for admin intervention)
5. Print summary at end: `{total: N, skipped: M, rewritten: K, failed: F}`. Exit 1 if `failed > 0`, else 0.

Critical: the script NEVER logs plaintext (`plaintext` is a local variable that goes out of scope after `encryptSecret` is called). The verbose mode logs only `{subdomain, providerId, action}` — no key material or envelope substrings.

      ```typescript
      #!/usr/bin/env tsx
      /**
       * ERXES_SECRET rotation script — D-03c (PITFALLS §P14 Policy A).
       *
       * Reads every ai_providers row across all subdomains, decrypts with whichever
       * version of ERXES_SECRET still works (Plan 01's multi-version decrypt), and
       * re-encrypts with the NEWEST version (encryptSecret always uses the first
       * entry of ERXES_SECRET).
       *
       * Idempotent: running on already-newest-version data is a no-op (envelope
       * bytes-equality check before write).
       *
       * Usage:
       *   ERXES_SECRET="v2:<new>,v1:<old>" MONGO_URL="..." pnpm tsx scripts/ai/rotate-secret.ts
       *   pnpm tsx scripts/ai/rotate-secret.ts --dry-run
       *   pnpm tsx scripts/ai/rotate-secret.ts --subdomain tenant-a --verbose
       *
       * Returns exit code 1 if any provider doc failed to decrypt under any
       * configured version (those rows need admin re-entry — see scripts/ai/README.md).
       */
      import mongoose from 'mongoose';
      import { encryptSecret, decryptSecret, SecretRotationRequiredError } from 'erxes-api-shared/ai';

      export interface RotationResult {
        total: number;
        skipped: number;
        rewritten: number;
        failed: number;
        durationMs: number;
      }

      export interface RotationOptions {
        dryRun?: boolean;
        subdomainFilter?: string;
        verbose?: boolean;
        db?: { listCollections(): Promise<Array<{name: string}>>; collection(name: string): { find(): { toArray(): Promise<any[]> }; updateOne(filter: any, update: any): Promise<any> } };
        encryptOverride?: typeof encryptSecret;
        decryptOverride?: typeof decryptSecret;
      }

      export const runRotation = async (opts: RotationOptions = {}): Promise<RotationResult> => {
        const start = Date.now();
        const dryRun = !!opts.dryRun;
        const verbose = !!opts.verbose;
        const enc = opts.encryptOverride ?? encryptSecret;
        const dec = opts.decryptOverride ?? decryptSecret;

        let db = opts.db;
        if (!db) {
          if (!process.env.MONGO_URL) {
            console.error('MONGO_URL not set; cannot rotate without database connection');
            process.exit(1);
          }
          await mongoose.connect(process.env.MONGO_URL);
          db = mongoose.connection.db as any;
        }

        const result: RotationResult = { total: 0, skipped: 0, rewritten: 0, failed: 0, durationMs: 0 };

        const collections = await db!.listCollections();
        const aiProviderCollections = collections.filter(c => /^[a-z0-9_-]+_ai_providers$/.test(c.name));

        for (const col of aiProviderCollections) {
          const match = col.name.match(/^([a-z0-9_-]+)_ai_providers$/);
          if (!match) continue;
          const subdomain = match[1];
          if (opts.subdomainFilter && opts.subdomainFilter !== subdomain) continue;

          const docs = await db!.collection(col.name).find().toArray();
          for (const doc of docs) {
            result.total += 1;
            try {
              const plaintext = dec(doc.encryptedApiKey, subdomain);
              const newEnvelope = enc(plaintext, subdomain);
              // plaintext local goes out of scope here

              if (newEnvelope === doc.encryptedApiKey) {
                result.skipped += 1;
                if (verbose) console.log(JSON.stringify({ subdomain, providerId: doc._id, action: 'skipped' }));
                continue;
              }

              if (!dryRun) {
                await db!.collection(col.name).updateOne({ _id: doc._id }, { $set: { encryptedApiKey: newEnvelope } });
              }
              result.rewritten += 1;
              if (verbose) console.log(JSON.stringify({ subdomain, providerId: doc._id, action: 'rewritten', dryRun }));
            } catch (e) {
              const isRotation = e instanceof SecretRotationRequiredError;
              result.failed += 1;
              console.log(JSON.stringify({
                subdomain,
                providerId: doc._id,
                action: 'failed',
                reason: isRotation ? 'SECRET_ROTATION_NEEDS_RESET' : 'UNKNOWN',
              }));
            }
          }
        }

        result.durationMs = Date.now() - start;
        console.log(JSON.stringify({ summary: result }));
        return result;
      };

      // Main entry — argv parsing
      const main = async () => {
        const argv = process.argv.slice(2);
        const dryRun = argv.includes('--dry-run');
        const verbose = argv.includes('--verbose');
        const subdomainIdx = argv.indexOf('--subdomain');
        const subdomainFilter = subdomainIdx >= 0 ? argv[subdomainIdx + 1] : undefined;

        const result = await runRotation({ dryRun, verbose, subdomainFilter });
        if (mongoose.connection.readyState === 1) await mongoose.disconnect();
        process.exit(result.failed > 0 ? 1 : 0);
      };

      if (require.main === module) {
        main().catch((e) => {
          console.error(JSON.stringify({ error: e instanceof Error ? e.message : 'unknown' }));
          process.exit(1);
        });
      }
      ```
